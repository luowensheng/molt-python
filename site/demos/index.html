def fetch_posts(limit: int = 10) -> list[dict]:
    url = "https://jsonplaceholder.typicode.com/posts"
    resp = requests.get(url, timeout=10)
    resp.raise_for_status()
    return resp.json()[:limit]

def show_posts(limit: int = 5) -> None:
    posts = fetch_posts(limit)
    table = Table(show_header=True,
                  header_style="bold magenta")
    table.add_column("ID", style="dim", width=4)
    table.add_column("Title")
    table.add_column("User", width=8)
    for post in posts:
        table.add_row(str(post["id"]),
                      post["title"][:60],
                      str(post["userId"]))
    console.print(table)

app = FastAPI(title="Task API", version="0.1.0")
_tasks: dict[int, dict] = {}

class TaskIn(BaseModel):
    title: str
    done: bool = False

@app.get("/tasks", response_model=list[Task])
def list_tasks():
    return list(_tasks.values())

@app.post("/tasks", response_model=Task, status_code=201)
def create_task(body: TaskIn):
    global _next_id
    task = {"id": _next_id, "title": body.title, "done": body.done}
    _tasks[_next_id] = task; _next_id += 1
    return task

@app.delete("/tasks/{task_id}", status_code=204)
def delete_task(task_id: int):
    if task_id not in _tasks:
        raise HTTPException(status_code=404, detail="Task not found")
    del _tasks[task_id]

Mojo matrix multiplication

Three implementations of N×N matrix multiplication in increasing order of optimization: naive O(N³) triple loop → SIMD-vectorized inner loop → cache-friendly tiled + parallelized. All three benchmarked side-by-side.

Layout

11-mojo-matmul/
├── main.mojo   # Matrix struct + 3 matmul implementations
└── pyproject.toml

Key code — three implementations

comptime NELTS = 4   # 128-bit NEON: 4 × float32 per register
comptime TILE  = 4   # tune to cache line

# 1. Naive O(N³)
def matmul_naive(mut C: Matrix, A: Matrix, B: Matrix):
    for m in range(C.rows):
        for k in range(A.cols):
            for n in range(C.cols):
                C[m, n] += A[m, k] * B[k, n]

# 2. SIMD-vectorized inner loop
def matmul_vectorized(mut C: Matrix, A: Matrix, B: Matrix):
    for m in range(C.rows):
        for k in range(A.cols):
            var a_mk = A[m, k]; var n = 0
            while n + NELTS <= C.cols:
                C.store[NELTS](m, n, C.load[NELTS](m,n) + a_mk * B.load[NELTS](k,n))
                n += NELTS

# 3. Tiled + parallelized
def matmul_tiled(mut C: Matrix, A: Matrix, B: Matrix):
    @parameter
    def calc_row(m: Int):
        for ko in range(0, A.cols, TILE):
            # ... tiled SIMD inner loop ...
    parallelize[calc_row](C.rows)

@export
def PyInit_fast_math() -> PythonObject:
    var m = PythonModuleBuilder("fast_math")
    m.def_function[factorial]("factorial",
        docstring="Compute n!")
    m.def_function[fast_sigmoid]("sigmoid",
        docstring="SIMD sigmoid over a list")
    m.def_function[dot_product]("dot",
        docstring="Dot product of two lists")
    _ = (m.add_type[RunningStats]("RunningStats")
           .def_py_init[RunningStats.py_init]()
           .def_method[RunningStats.py_update]("update")
           .def_method[RunningStats.py_mean]("mean"))
    return m.finalize()

⚙️ C extension Python ↔ C zero boilerplate

C extension module — header-driven, bidirectional, 30–100× faster

First-class C extensions with zero manifest boilerplate. Declare [[tool.molt.c.modules]] in pyproject.toml, point at a .h header, and molt sync auto-parses every scalar function signature, generates a full CPython PyInit_ / PyMethodDef module (not ctypes!), and compiles via zig cc. The result is a real native CPython extension: 30–100× faster than Python for math-heavy loops, because values are never boxed into Python objects. The relationship is bidirectional: Python calls C at native speed, and C can call back into Python packages via PyImport_ImportModule — numpy, scipy, pandas are all reachable because PYTHONPATH is set to the full molt package store.

struct Stats {
    n: usize, sum: f64, mean: f64,
    std_dev: f64, min: f64, max: f64,
}

fn compute(data: &[f64]) -> Option<Stats> {
    if data.is_empty() { return None; }
    let sum: f64 = data.iter().sum();
    let mean = sum / data.len() as f64;
    let variance: f64 = data.iter()
        .map(|&x| { let d = x - mean; d * d })
        .sum::<f64>() / data.len() as f64;
    Some(Stats { n: data.len(), sum, mean,
                 std_dev: variance.sqrt(), .. })
}

// Mean returns the arithmetic mean of data.
func Mean(data []float64) float64 {
    sum := 0.0
    for _, v := range data { sum += v }
    return sum / float64(len(data))
}

func Stddev(data []float64) float64 {
    m := Mean(data); v := 0.0
    for _, x := range data {
        d := x - m; v += d * d
    }
    return math.Sqrt(v / float64(len(data)))
}

pub fn deflate(data: Vec<u8>) -> Vec<u8> {
    let mut enc = ZlibEncoder::new(
        Vec::new(), Compression::default());
    enc.write_all(&data).unwrap_or(());
    enc.finish().unwrap_or_default()
}

pub fn inflate(data: Vec<u8>) -> Vec<u8> {
    let mut dec = ZlibDecoder::new(data.as_slice());
    let mut out = Vec::new();
    dec.read_to_end(&mut out).unwrap_or(0); out
}
